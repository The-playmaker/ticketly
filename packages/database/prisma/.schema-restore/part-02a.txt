
  @@index([eventId, status, createdAt])
  @@index([eventId, email])
  @@index([ticketTypeId, status])
  @@index([inviteToken])

enum ReminderKind {
  EVENT_24H
  EVENT_3H
  EVENT_CHANGED
  EVENT_CANCELLED
}

enum ReminderJobStatus {
  SCHEDULED
  SENT
  FAILED
  CANCELLED
  SKIPPED
}

model EventReminderJob {
  id           String            @id @default(cuid())
  eventId      String
  orderId      String
  kind         ReminderKind
  scheduledFor DateTime
  status       ReminderJobStatus @default(SCHEDULED)
  sentAt       DateTime?
  error        String?
  createdAt    DateTime          @default(now())
  updatedAt    DateTime          @updatedAt

  event Event @relation(fields: [eventId], references: [id], onDelete: Cascade)

  @@unique([eventId, orderId, kind])
  @@index([status, scheduledFor])
  @@index([eventId, kind])
}

/// Early-bird / tiered pricing: first N tickets at price X, next M at Y, etc.
