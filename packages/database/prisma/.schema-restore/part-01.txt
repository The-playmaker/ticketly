  slug                 String

  category             String?

  salesEndAt           DateTime?
  publishAt            DateTime?
  /// Emergency / manual pause — blocks NEW reservations only
  salesPausedAt        DateTime?
  availabilityDisplay  AvailabilityDisplayMode @default(SHOW_LOW_STOCK_ONLY)
  lowStockThreshold    Int             @default(10)
  soldOutDisplay       SoldOutDisplayMode @default(SOLD_OUT)

  dynamicQrTtlSeconds  Int             @default(30)

  waitlistOnReleaseSoldOut Boolean     @default(false)
  /// Transactional event reminders (not marketing)
  reminder24hEnabled   Boolean         @default(true)
  reminder3hEnabled    Boolean         @default(false)
  seriesId             String?
  createdAt            DateTime        @default(now())
  updatedAt            DateTime        @updatedAt

  sellerOrganization Organization              @relation("EventSeller", fields: [sellerOrganizationId], references: [id], onDelete: Restrict)

  assignments        EventAssignment[]

  offlineCheckIns    OfflineCheckIn[]
  analyticsEvents    AnalyticsEvent[]
  waitlistEntries    WaitlistEntry[]
  ticketBatches      TicketBatch[]
  reminderJobs       EventReminderJob[]

  @@unique([organizationId, slug])
  @@index([slug])
  @@index([status])
  @@index([startsAt])
  @@index([organizationId, status])
  @@index([sellerOrganizationId])
  @@index([visibility, status, startsAt])
  @@index([seriesId])
}
